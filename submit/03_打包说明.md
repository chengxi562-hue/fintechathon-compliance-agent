## 方法 B：Python（本机无 zip CLI 时用此法；初稿已用此法生成）

```bash
cd <repo-root>
python3 << 'PY'
import zipfile
from pathlib import Path
root = Path(".").resolve()
out = root / "submit" / "03_源代码与部署_v1.zip"
SKIP_DIR = {".venv","venv","__pycache__",".pytest_cache",".mypy_cache",".ruff_cache",".git",".idea",".vscode",".refs","_build"}
SKIP_NAME = {".DS_Store",".env"}
def skip(rel: Path) -> bool:
    if any(p in SKIP_DIR for p in rel.parts): return True
    if rel.name in SKIP_NAME: return True
    if rel.name.startswith(".env.") and rel.name != ".env.example": return True
    if rel.suffix in {".pyc",".pyo",".log"}: return True
    if rel.name.endswith(".zip") and "submit" in rel.parts: return True
    if rel.parts[:2] == ("data","audit") and rel.name.endswith(".jsonl"): return True
    return False
with zipfile.ZipFile(out, "w", compression=zipfile.ZIP_DEFLATED) as zf:
    for p in root.rglob("*"):
        if p.is_file() and not skip(p.relative_to(root)):
            zf.write(p, (Path("fintechathon") / p.relative_to(root)).as_posix())
print("wrote", out, out.stat().st_size)
PY
```
